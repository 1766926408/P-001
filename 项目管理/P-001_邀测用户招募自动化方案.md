### 模块 2: 微信群机器人（自动化）

**工具链**：使用 **WeChatBot** + **接收自动响应**

```python
# wechat_bot.py - 自动化脚本

import itchat
from itchat.content import TEXT, IMAGE
import gspread
from google.oauth2.service_account import Credentials
import time

# 连接Google Sheets
gc = gspread.service_account(filename='service_account.json')
sheet = gc.open('user-base-tracking').sheet1

# 自动回复模板
AUTO_REPLY = """
欢迎加入【产品名称】邀测群！🎉

👇 请按下列步骤完成登记：
1️⃣ 点击链接，填写用户信息：[邀测链接]
2️⃣ 每周五提交一份反馈
3️⃣ 邀请2位用户可获得奖励积分

有问题？回复【问题】获取帮助
"""

@itchat.msg_register(TEXT, isFriendChat=True)
def reply_friend(msg):
    # 自动回复邀请信息
    if '邀请' in msg['Text'] or '招募' in msg['Text']:
        itchat.send(AUTO_REPLY, msg['FromUserName'])
        # 自动记录到Sheet
        add_user_to_sheet(msg['FromUserName'], 'wechat')

@itchat.msg_register(TEXT, isGroupChat=True)
def reply_group(msg):
    # 监听群内反馈
    if '反馈' in msg['Text'] or '@' in msg['Text']:
        # 自动抓取反馈到Sheet
        add_feedback_to_sheet(msg['FromUserName'], msg['Text'], 'wechat')

def add_user_to_sheet(user_id, source):
    """自动添加用户到Sheet"""
    sheet.append_row([
        uuid.uuid4(),
        user_id,
        source,
        generate_invite_link(),
        datetime.now().isoformat(),
        '已加入'
    ])

def add_feedback_to_sheet(user_id, feedback, source):
    """自动记录反馈"""
    feedback_sheet = gc.open('user-base-tracking').worksheet('feedback-log')
    feedback_sheet.append_row([source, user_id, feedback, datetime.now().isoformat(), '未读'])

itchat.auto_login(hotReload=True)
itchat.run()
```

### 模块 3: QQ群机器人（自动化）

**工具链**：使用 **go-cqhttp** + **nonebot2**

```python
# qq_bot.py - QQ群自动邀请脚本

from nonebot import on_message, on_command
from nonebot.adapters import cqhttp
import aiohttp

matcher = on_message()

@matcher.handle()
async def handle_msg(bot: cqhttp.Bot, event: cqhttp.GroupMessageEvent):
    """监听群消息，自动追踪用户"""
    if event.raw_message and ('反馈' in event.raw_message or '建议' in event.raw_message):
        # 自动发送感谢 + 激励提示
        await bot.send(event, "感谢反馈！已记录，您获得10积分 🎁")
        # 同步到Google Sheets
        await sync_qq_feedback(event.user_id, event.raw_message)

async def sync_qq_feedback(user_id, content):
    """异步同步到Google Sheets"""
    # 通过API调用Google Apps Script
    async with aiohttp.ClientSession() as session:
        await session.post(
            'https://script.google.com/macros/d/{SCRIPT_ID}/usercontent',
            json={'user_id': user_id, 'content': content, 'source': 'qq'}
        )
```

### 模块 6: 激励自动计算 & 发放

**触发机制**：每天早6点自动计算，9点前发放

```python
# auto_reward.py - 激励系统

import gspread
from datetime import datetime

def calculate_rewards():
    """自动计算激励"""
    sheet = gc.open('user-base-tracking')
    users = sheet.worksheet('users-base').get_all_records()
    
    for user in users:
        points = 0
        
        # 规则1：加入群聊 +5积分
        if user['群聊状态'] == '已加入':
            points += 5
        
        # 规则2：提交反馈 +10积分
        if user['群聊状态'] == '已反馈':
            points += 10
        
        # 规则3：邀请朋友 +15积分/人
        points += user['邀请数量'] * 15
        
        # 更新积分
        update_user_points(user['用户ID'], points)

def auto_reward_distribution():
    """自动发放激励"""
    rewards = calculate_rewards()
    
    # 向微信/QQ发送激励通知
    for user_id, reward in rewards.items():
        send_reward_notification(user_id, reward)
    
    # 更新发放状态
    mark_rewards_sent()

# 使用APScheduler定时触发
from apscheduler.schedulers.background import BackgroundScheduler
scheduler = BackgroundScheduler()
scheduler.add_job(calculate_rewards, 'cron', hour=6)  # 每天早6点
scheduler.add_job(auto_reward_distribution, 'cron', hour=9)  # 每天早9点
scheduler.start()
```
