```python
# 脚本：sync_form_to_db.py
# 功能：每天自动从问卷星导出数据 → SQLite

import pandas as pd
from modules.sqlite_manager import SQLiteManager

def sync_daily():
    # 1. 从问卷星导出CSV
    df = pd.read_csv('wjx_export.csv')
    
    # 2. 同步到SQLite
    db = SQLiteManager()
    for _, row in df.iterrows():
        db.add_user(
            nickname=row['姓名'],
            source='form',
            invite_link=f"https://invite.motormind.com/{row['姓名']}"
        )
        db.add_feedback(
            source='form',
            user_name=row['姓名'],
            content=row['期望反馈重点']
        )
    
    db.close()
    print("✅ 数据同步完成")

# 每天早10点自动执行
```

```python
# 脚本：auto_feedback_classify.py

def classify_feedback(content):
    """自动分类用户反馈"""
    
    categories = {
        'bug': ['错误', 'bug', '崩溃', '异常'],
        'feature': ['功能', '建议', '改进'],
        'ui': ['界面', 'UI', '样式', '美观'],
        'performance': ['性能', '慢', '卡顿', '响应']
    }
    
    for cat, keywords in categories.items():
        if any(kw in content for kw in keywords):
            return cat
    
    return 'other'

# 自动分类所有反馈
```

## 🔧 部署方案

### 方案 A：本地 + GitHub Actions（推荐）

```
使用：GitHub Actions + GitHub Secrets

配置步骤：
1. 创建 .github/workflows/auto_tasks.yml
2. 设置 GOOGLE_SHEET_ID 等密钥
3. 配置执行时间表（cron）
4. Push 到 GitHub
5. 自动执行（24/7）

优点：
  ✓ 完全免费
  ✓ 24/7 自动执行
  ✓ 无需本地机器
  ✓ 版本控制
```

**GitHub Actions 工作流示例**：

```yaml
name: MotorMind Auto Tasks

on:
  schedule:
    - cron: '0 6 * * *'  # 早6点 - 计算激励
    - cron: '0 9 * * *'  # 早9点 - 发放激励
    - cron: '0 10 * * *' # 早10点 - 同步数据
    - cron: '0 17 * * 5' # 周五5点 - 生成周报

jobs:
  tasks:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v2
      
      - name: Set up Python
        uses: actions/setup-python@v2
        with:
          python-version: '3.9'
      
      - name: Install dependencies
        run: |
          pip install -r requirements.txt
      
      - name: Run tasks
        env:
          SQLITE_DB: recruitment.db
        run: |
          python main.py
```
